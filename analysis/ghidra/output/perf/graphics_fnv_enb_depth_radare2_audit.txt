allowed = [(0x108F10C0, 0x108F11EE), (0x108F1236, 0x108F1267)]

def check_block(emulator, address, size, _):
    if not any(start <= address < end for start, end in allowed):
        raise RuntimeError(f'Unapproved stub block at {address:#x}')

uc.hook_add(UC_HOOK_BLOCK, check_block)
uc.emu_start(0x108F10C0, 0x108F11EE, timeout=60_000_000, count=200_000_000)
assert uc.reg_read(UC_X86_REG_EIP) == 0x108F11EE, 'Incomplete decompression'
memory = bytearray(uc.mem_read(BASE, SIZE))

# The actual stub at 108F11EE resolves this compact import table. Read it
# without invoking LoadLibrary/GetProcAddress, then resume the actual
# relocation instructions with EDI at the same terminating table entry.
cursor = 0x8EB000
imports = []
def cstring(buf, offset):
    end = buf.index(0, offset)
    assert end - offset < 512
    return bytes(buf[offset:end]).decode('ascii'), end + 1

while u32(memory, cursor):
    name_offset, iat_offset = struct.unpack_from('<II', memory, cursor)
    library, _ = cstring(memory, name_offset + 0x8F2558)
    cursor += 8
    names = []
    while memory[cursor]:
        marker = memory[cursor]
        cursor += 1
        assert marker == 1, f'Unproven import marker {marker}'
        name, cursor = cstring(memory, cursor)
        names.append(name)
    cursor += 1
    imports.append({'library': library, 'iat_rva': 0x1000 + iat_offset, 'names': names})

uc.reg_write(UC_X86_REG_EDI, BASE + cursor)
uc.emu_start(0x108F1236, 0x108F1267, timeout=60_000_000, count=20_000_000)
assert uc.reg_read(UC_X86_REG_EIP) == 0x108F1267, 'Incomplete relocation'
memory = bytearray(uc.mem_read(BASE, SIZE))
Path('/tmp/omv_enb_unpacked_memory.bin').write_bytes(memory)
Path('/tmp/omv_enb_imports.json').write_text(json.dumps(imports, indent=2) + '\n')

# Synthesize a single RVA=file-offset section and a conventional import
# directory for radare2. All original executable bytes retain their RVAs.
directory_rva = SIZE
analysis = memory + bytearray((len(imports) + 1) * 20)
def append(data, alignment=1):
    analysis.extend(bytes((-len(analysis)) % alignment))
    rva = len(analysis)
    analysis.extend(data)
    return rva

for index, library in enumerate(imports):
    name_rva = append(library['library'].encode() + b'\0')
    names = [append(b'\0\0' + name.encode() + b'\0', 2) for name in library['names']]
    thunk = b''.join(struct.pack('<I', x) for x in names) + bytes(4)
    ilt_rva = append(thunk, 4)
    iat = library['iat_rva']
    analysis[iat:iat + len(thunk)] = thunk
    struct.pack_into('<IIIII', analysis, directory_rva + index * 20,
                     ilt_rva, 0, 0, name_rva, iat)

analysis.extend(bytes((-len(analysis)) % 0x1000))
