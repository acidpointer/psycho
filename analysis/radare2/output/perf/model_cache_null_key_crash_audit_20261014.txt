        && let Some(expanded) = capacity.checked_add(1)
    {
        return expanded;

        log::error!(
            "[IO] Invalid LockFreeMap TLS capacity {} in family {} count={count}",
            capacity,
            family + 1,
        );
    }
    capacity
}
